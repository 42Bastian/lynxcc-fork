; Format Notes:
;
; sfx_name priority
;	waveform feedback [shift]
;	volume amplitude [adjustment]
;	frequency divider [adjustment]
;	tuning value
;	{
;		waveform feedback [shift]
;		volume amplitude [adjustment]
;		frequency divider [adjustment]
;		loop count
;			rest delay
;		endloop
;		noteoff
;		end
;	}
;

; Zaku_RapidShot 128
;	Zaku's Rapid shot
;**************************************
Zaku_RapidShot 128

	volume 30 -5
	frequency $2E $FFFF.70
	{
		rest 4
		end
	}

;**************************************
; Zaku_ChargeUp 132
;	Zaku's Charge Up
;**************************************
Zaku_ChargeUp 132

		end
	}
