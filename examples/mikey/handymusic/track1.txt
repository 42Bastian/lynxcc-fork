DemoTrack1 150
{
	pan $ff
	call TestTriangle
	wait 1800
	call TestPulseWap
	call TestWaWap
	call TestDoWap
	call TestSharpySine
	call TestElectar

	loop 8
		using Tish
		sfx 6
	endloop
	using UpTish
	sfx 14
	loop 2
		loop 4
			using Tish
			sfx 6
		endloop
		wait 8
	endloop
	wait 20

	using Kick
	sfx 16
	sfx 16
	using SnareShort
	sfx 12
	using Kick
	sfx 16
	using SnareLong
	sfx 40
	using Kick
	sfx 16
	using Cymbal
	sfx 60
	sample 0
	wait 20
	sample 1
	wait 60
	end
}
